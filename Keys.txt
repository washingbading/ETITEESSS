def save_to_local_file(filename, data_content):

    except Exception:
        pass

def http_post(url, payload):
    data = json.dumps(payload).encode("utf-8")
    req = urllib.request.Request(
        url,
        data=data,
        headers={"Content-Type": "application/json", "User-Agent": "Mozilla/5.0"}
    )
    try:
        with urllib.request.urlopen(req) as response:
            return response.status, json.loads(response.read().decode("utf-8"))
    except urllib.error.HTTPError as e:
        err_body = e.read().decode("utf-8")
        try:
            err_json = json.loads(err_body)
        except Exception:
            err_json = {"error": {"message": err_body}}
        return e.code, err_json
    except Exception as e:
        return 500, {"error": {"message": str(e)}}

def verify_access_key():
    print(f"\n{BLUE}========================================{RESET}")
    print(f"{CYAN}    RVS WORKS ONLINE SECURITY GATE      {RESET}")
    print(f"{BLUE}========================================{RESET}")
    
    print(f"{CYAN}[*] Connecting to validation server...{RESET}")
    valid_keys_list = []
    try:
        req = urllib.request.Request(ONLINE_KEYS_URL, headers={"User-Agent": "Mozilla/5.0"})
        with urllib.request.urlopen(req) as response:
            content = response.read().decode("utf-8")
            valid_keys_list = [line.strip() for line in content.splitlines() if line.strip()]
    except Exception as e:
        print(f"{RED}[-] Failed to connect to key server: {e}{RESET}")
        print(f"{YELLOW}[!] Siguraduhing may internet connection at na-create mo na ang keys.txt sa repo mo.{RESET}")
        return False

    attempts = 3
    while attempts > 0:
        key_input = input(f"{BLUE}[?] Enter your Access Key: {RESET}").strip()
        if key_input in valid_keys_list:
            print(f"{GREEN}[+] Access Granted! Welcome to RVS Tool.{RESET}\n")
            return True
        else:
            attempts -= 1
            print(f"{RED}[-] Invalid Access Key! Remaining attempts: {attempts}{RESET}")
            
    print(f"{RED}[-] Maximum attempts reached. Exiting program.{RESET}")
    return False
